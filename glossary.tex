\newglossaryentry{syscall}{
	name=System Call,
	description={A low-level function call used by user-mode programs to interact with the kernel of the underlying operating system, usually to interact with the system's hardware.}
}

\newglossaryentry{tribalknowledge}{
	name=tribal knowledge,
	description={Knowledge which is not documented through permanent means such as documentation files or wikis, but is rather stored in the minds of people handling the knowledge (the \enquote{tribe}).}
}

\newglossaryentry{reconcile}{
	name=reconcile,
	description={To move the observed state of a system closer to the desired state by running a series of actions that may or may not achieve that goal at the end of the reconciliation.
			Reconciliation usually runs in a loop.
		}
}

\newglossaryentry{idempotency}{
	name=idempotency,
	description={A property of an action, program or similar that describes that the action will have the same result, whether it is executed once or multiple times, and will not mutate the state of the subject if the subject is already in the target state.}
}

\newglossaryentry{portforward}{
    name=port forwarding,
    description={Performing a destination network address translation that forwards a packet to a different host than the original destination, commonly used to forward packets arriving at a single \gls{ip} address to several different hosts on an internal network.}
}
